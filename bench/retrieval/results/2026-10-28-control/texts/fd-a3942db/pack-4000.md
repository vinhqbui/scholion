```rust
pub fn sanitize_for_terminal(s: &str) -> Cow<'_, str> {
    if !s.chars().any(needs_escape) {
        return Cow::Borrowed(s);
    }
    let mut out = String::with_capacity(s.len());
    for c in s.chars() {
        if needs_escape(c) {
            let v = c as u32;
            if v <= 0xFF {
                let _ = write!(out, "\\x{v:02X}");
            } else {
                let _ = write!(out, "\\u{{{v:04X}}}");
            }
        } else {
            out.push(c);
        }
    }
    Cow::Owned(out)
}
```
